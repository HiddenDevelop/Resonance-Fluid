cmake_minimum_required(VERSION 3.24)
# Set before project(), which initializes architecture defaults during CUDA detection.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native)
endif()
project(Resonance VERSION 0.1.0 LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)

find_package(CUDAToolkit 12.0 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import jinja2"
    RESULT_VARIABLE JINJA_RESULT OUTPUT_QUIET ERROR_QUIET)
if(NOT JINJA_RESULT EQUAL 0)
    message(FATAL_ERROR "GLAD needs Jinja2. Run: ${Python3_EXECUTABLE} -m pip install Jinja2==3.1.6")
endif()
include(FetchContent)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git
    GIT_TAG a74efa0d5628b74adc0426af4c5710e287fa7c2c)
FetchContent_Declare(glad GIT_REPOSITORY https://github.com/Dav1dde/glad.git
    GIT_TAG 73db193f853e2ee079bf3ca8a64aa2eaf6459043 SOURCE_SUBDIR cmake)
FetchContent_MakeAvailable(glfw glad)
glad_add_library(glad_gl STATIC REPRODUCIBLE API gl:core=3.3)
add_library(resonance_fluid STATIC src/FluidSimulation.cu)
target_include_directories(resonance_fluid PUBLIC src)
target_link_libraries(resonance_fluid PUBLIC CUDA::cudart)
target_compile_options(resonance_fluid PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
add_executable(Resonance src/main.cpp src/Renderer.cpp src/Input.cpp)
target_compile_definitions(Resonance PRIVATE GLFW_INCLUDE_NONE NOMINMAX)
target_link_libraries(Resonance PRIVATE resonance_fluid glad_gl glfw OpenGL::GL)
if(MSVC)
    target_compile_options(Resonance PRIVATE /W4 /permissive-)
endif()
add_custom_command(TARGET Resonance POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:Resonance>/shaders")
include(CTest)
if(BUILD_TESTING)
    add_executable(fluid_tests tests/fluid_tests.cu)
    target_link_libraries(fluid_tests PRIVATE resonance_fluid)
    add_test(NAME fluid_gpu COMMAND fluid_tests)
    set_tests_properties(fluid_gpu PROPERTIES SKIP_RETURN_CODE 77)
endif()
